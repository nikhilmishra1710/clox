    "name": "Alice",

};

user["role"] = "owner";
user[" status"] = "active";

print user.has("name");  // true

        print a * 10 + b;
        b = b + 1;

    a = a + 1;
}

var x = 10;
x += 5;
print x;      // Expect: 15

x -= 3;
print x;      // Expect: 12

x *= 2;
print x;      // Expect: 24

x /= 4;
print x;      // Expect: 6

print x++;    // Expect: 6 (postfix returns old value)
print x;      // Expect: 7

print ++x;    // Expect: 8 (prefix returns new value)
print --x;    // Expect: 7

// Works in for loops!
for (var i = 0; i < 3; i++) {
    print i;  // Expect: 0, 1, 2
}
