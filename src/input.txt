// 2. Stack unwinding across multiple call frames

    throw "deep error";
}

    var local = "cleaned up";

    level1();
} catch (err) {
    print "2. Unwound: " + err; // Expect: 2. Unwound: deep error

// 3. 'continue' and 'break' inside a try block inside a loop
// (Tests that OP_POP_HANDLER is emitted properly so handler stack doesn't leak!)
try {
    for (var i = 0; i < 5; i++) {
        try {
            if (i == 1) continue;
            if (i == 3) break;
            print "3. Loop i = ${i}";
        } catch (innerErr) {
            print "FAIL: inner catch should not run";
        }
    }
    // Now throw outside the loop: should be caught by OUTER try, NOT the broken inner try!
    throw "outer catch works!";
} catch (outerErr) {
    print "3. " + outerErr;
}
// Expect:
// 3. Loop i = 0
// 3. Loop i = 2
// 3. outer catch works!

// 2. Closure over local variable
fun makeCounter() {
    var count = 0;
    return fun() {
        count += 1;
        return count;
    };
}
var c = makeCounter();
print c(); // Expect: 1
print c(); // Expect: 2

// 3. Passing a lambda as a callback to a higher-order function
fun map(list, fn) {
    var out = [];
    for (var i = 0; i < list.len(); i++) {
        out.push(fn(list[i]));
    }
    return out;
}

var nums = [1, 2, 3, 4];
var squared = map(nums, fun(x) { return x * x; });
print squared; // Expect: [1, 4, 9, 16]

// 4. Immediately Invoked Function Expression (IIFE)
