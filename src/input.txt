var a = [10, 20, 30];
print a;          // Expect: [10, 20, 30]
print a[0];       // Expect: 10
a[1] = 99;
a.push(40);
print a;          // Expect: [10, 99, 30, 40]
print a.pop();    // Expect: 40
print a.len();    // Expect: 3
